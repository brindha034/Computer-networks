#include <stdio.h>
#include <string.h>

int main()
{
    char data[100], encoded[200], received[200];
    int i, j, k, n, r, pos;
    int parity, error = 0;

    printf("Enter binary data: ");
    scanf("%s", data);

    n = strlen(data);

    r = 0;
    while ((1 << r) < n + r + 1)
        r++;

    int total = n + r;

    j = 0;

    for (i = 1; i <= total; i++)
    {
        if ((i & (i - 1)) == 0)
            encoded[i] = '0';
        else
        {
            encoded[i] = data[j];
            j++;
        }
    }

    for (i = 0; i < r; i++)
    {
        pos = 1 << i;
        parity = 0;

        for (j = 1; j <= total; j++)
        {
            if (j & pos)
            {
                if (encoded[j] == '1')
                    parity++;
            }
        }

        encoded[pos] = (parity % 2) + '0';
    }

    FILE *fp = fopen("channel.txt", "w");

    for (i = 1; i <= total; i++)
        fputc(encoded[i], fp);

    fclose(fp);

    printf("\nHamming Code Sent: ");

    for (i = 1; i <= total; i++)
        printf("%c", encoded[i]);

    printf("\n\nEnter received Hamming code: ");
    scanf("%s", received);

    error = 0;

    for (i = 0; i < r; i++)
    {
        pos = 1 << i;
        parity = 0;

        for (j = 1; j <= total; j++)
        {
            if (j & pos)
            {
                if (received[j - 1] == '1')
                    parity++;
            }
        }

        if (parity % 2 != 0)
            error += pos;
    }

    if (error == 0)
    {
        printf("\nNo error detected.\n");
    }
    else
    {
        printf("\nError detected at position: %d\n", error);

        if (received[error - 1] == '0')
            received[error - 1] = '1';
        else
            received[error - 1] = '0';

        printf("Corrected Hamming Code: %s\n", received);
    }

    return 0;
}
